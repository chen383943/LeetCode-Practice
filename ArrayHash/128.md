class Solution:
    def longestConsecutive(self, nums: list[int]) -> int:
        num_set = set(nums)
        longest = 0
        for num in num_set:  # Iterate directly over set to reduce duplicate checks
            if num - 1 not in num_set:  # Start of a sequence
                length = 1
                while num + 1 in num_set:  # Extend the sequence
                    num += 1
                    length += 1
                longest = max(longest, length)  # Update max length
        return longest
