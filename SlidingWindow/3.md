class Solution:
    def lengthOfLongestSubstring(self, s: str) -> int:
        start = 0
        res = 0
        unique = set()
        for end in range(len(s)):
            while s[end] in unique: #duplicate, update the window
                unique.remove(s[start]) 
                start += 1
            unique.add(s[end])
            res = max(res, end - start + 1)
        return res
